    ANIMAL_CONFIDENCE_THRESHOLD: float = 0.5
    BIRD_CONFIDENCE_THRESHOLD: float = 0.4
    
    # Performance settings (OPTIMIZED)
    MAX_FPS: int = 15  # Reduced for better performance
    DETECTION_FPS: int = 5  # Run detection at 5 FPS

    # Camera settings (OPTIMIZED)
    CAMERA_RESOLUTION: Tuple[int, int] = (640, 480)  # Lower resolution
    DETECTION_RESOLUTION: Tuple[int, int] = (416, 416)  # YOLO optimal size
    
    # Processing optimization
    SKIP_FRAMES: int = 0  # Don't skip, control via FPS
    MAX_DETECTIONS: int = 20  # Limit max detections per frame

    USE_FAST_RENDERING: bool = True  # Fast pygame rendering
    
    # Colors
    DETECTION_COLOR_HUMAN: Tuple[int, int, int] = (0, 255, 0)
    DETECTION_COLOR_ANIMAL: Tuple[int, int, int] = (0, 165, 255)
    DETECTION_COLOR_BIRD: Tuple[int, int, int] = (255, 0, 255)
    THREAT_COLOR_CRITICAL: Tuple[int, int, int] = (255, 0, 0)
    THREAT_COLOR_HIGH: Tuple[int, int, int] = (255, 165, 0)
    THREAT_COLOR_MEDIUM: Tuple[int, int, int] = (255, 255, 0)

@dataclass
class DetectionEvent:
    """Detection event data"""
    timestamp: datetime
    object_type: str
    category: str
    confidence: float
    bbox: Tuple[int, int, int, int]
    distance: float
    threat_level: str

class OptimizedCamera:
    """Optimized camera controller"""
    
    def __init__(self, config: SecurityConfig):
        self.config = config
        self.cap = None
        self.current_frame = None
        self.frame_lock = threading.Lock()
        self.running = False
        self.frame_count = 0
        self.fps_deque = deque(maxlen=30)
        self.last_time = time.time()
        
        logger.info("Camera controller initialized")
    
    def initialize(self, source: str = "webcam", **kwargs) -> bool:
        """Initialize camera"""
        try:
            if source == "webcam":
                camera_index = kwargs.get('camera_index', 0)
                logger.info(f"Opening webcam {camera_index}")
                
                # Try different backends
                for backend in [cv2.CAP_DSHOW, cv2.CAP_MSMF, cv2.CAP_ANY]:
                    self.cap = cv2.VideoCapture(camera_index, backend)
                    if self.cap.isOpened():
                        logger.info(f"Camera opened with backend {backend}")
                        break
                    self.cap.release()
                
                if not self.cap.isOpened():
                    logger.error("Failed to open camera")
                    return False
                
                # Set resolution
                self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, self.config.CAMERA_RESOLUTION[0])
                self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, self.config.CAMERA_RESOLUTION[1])
                self.cap.set(cv2.CAP_PROP_FPS, 30)
                self.cap.set(cv2.CAP_PROP_BUFFERSIZE, 1)  # Minimize buffer
                
            elif source == "video":
                video_path = kwargs.get('video_path')
                if not video_path or not Path(video_path).exists():
                    logger.error(f"Video file not found: {video_path}")
                    return False
                
                self.cap = cv2.VideoCapture(video_path)
                if not self.cap.isOpened():
                    logger.error(f"Failed to open video: {video_path}")
                    return False
            
            else:
                logger.error(f"Unknown source: {source}")
                return False
            
            # Test frame read
            ret, frame = self.cap.read()
            if not ret:
                logger.error("Failed to read test frame")
                return False
            
            logger.info(f"Camera initialized: {frame.shape[1]}x{frame.shape[0]}")
            return True
            
        except Exception as e:
            logger.error(f"Camera initialization error: {e}")
            return False
    
    def start_capture(self):
        """Start capture thread"""
        self.running = True
        self.thread = threading.Thread(target=self._capture_loop, daemon=True)
        self.thread.start()
        logger.info("Capture started")
    
    def _capture_loop(self):
        """Optimized capture loop"""
        frame_delay = 1.0 / self.config.MAX_FPS
        
        while self.running:
            loop_start = time.time()
            
            ret, frame = self.cap.read()
            if ret and frame is not None:
                with self.frame_lock:
                    self.current_frame = frame
                    self.frame_count += 1
                    
                    # Calculate FPS
                    current_time = time.time()
                    time_diff = current_time - self.last_time
                    if time_diff > 0:
                        self.fps_deque.append(1.0 / time_diff)
                    self.last_time = current_time
            
            # Frame rate limiting
            elapsed = time.time() - loop_start
            sleep_time = frame_delay - elapsed
            if sleep_time > 0:
                time.sleep(sleep_time)
    
    def get_frame(self) -> Optional[np.ndarray]:
        """Get current frame (no copy)"""
        with self.frame_lock:
            return self.current_frame
    
    def get_fps(self) -> float:
        """Get current FPS"""
        return np.mean(self.fps_deque) if self.fps_deque else 0.0
    
    def stop(self):
        """Stop capture"""
        self.running = False
        if hasattr(self, 'thread'):
            self.thread.join(timeout=1.0)
    
    def release(self):
        """Release camera"""
        if self.cap:
            self.cap.release()
        logger.info("Camera released")

class FastDashboard:
    """Optimized dashboard with fast rendering"""
    
    def __init__(self, config: SecurityConfig):
        self.config = config
        self.running = False
        
        if not PYGAME_AVAILABLE:
            logger.error("Pygame not available")
            return
        
        try:
            pygame.init()
            
            # Calculate display size
            self.display_width = int(config.CAMERA_RESOLUTION[0] * config.DISPLAY_SCALE)
            self.display_height = int(config.CAMERA_RESOLUTION[1] * config.DISPLAY_SCALE)
            self.panel_width = 350
            
            total_width = self.display_width + self.panel_width
            total_height = self.display_height
            
            # Create window
            self.screen = pygame.display.set_mode((total_width, total_height))
            pygame.display.set_caption("Security System - Optimized")
            self.clock = pygame.time.Clock()
            
            # Colors
            self.colors = {
                'background': (15, 15, 25),
                'panel': (25, 25, 40),
                'text': (220, 220, 240),
                'accent': (0, 180, 255),
                'success': (0, 255, 120),
                'warning': (255, 180, 0),
                'danger': (255, 60, 60)
            }
            
            # Fonts
            self.font_large = pygame.font.Font(None, 28)
            self.font_medium = pygame.font.Font(None, 22)
            self.font_small = pygame.font.Font(None, 18)
            
            # Frame surface cache
            self.frame_surface = None
            
            self.running = True
            logger.info(f"Dashboard initialized: {total_width}x{total_height}")
            
        except Exception as e:
            logger.error(f"Dashboard init error: {e}")
            self.running = False
    
    def update(self, frame: np.ndarray, detections: List, stats: Dict):
        """Fast update with optimized rendering"""
        if not self.running:
            return
        
        try:
            # Clear screen
            self.screen.fill(self.colors['background'])
            
            # Draw frame (OPTIMIZED)
            if frame is not None:
                self._draw_frame_fast(frame)
            
            # Draw info panel
            self._draw_info_panel(detections, stats)
            
            # Update display
            pygame.display.flip()
            self.clock.tick(self.config.UI_FPS)
            
        except Exception as e:
            logger.error(f"Dashboard update error: {e}")
    
    def _draw_frame_fast(self, frame: np.ndarray):
        """Optimized frame rendering"""
        try:
            # Resize once if needed
            if frame.shape[:2] != (self.display_height, self.display_width):
                frame = cv2.resize(frame, (self.display_width, self.display_height), 
                                 interpolation=cv2.INTER_LINEAR)
            
            # Convert BGR to RGB
            frame_rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB)
            
            # Convert to pygame surface (OPTIMIZED)
            # Transpose to get correct orientation
            frame_transposed = np.transpose(frame_rgb, (1, 0, 2))
            
            # Create or update surface
            if self.frame_surface is None or \
               self.frame_surface.get_size() != (self.display_width, self.display_height):
                self.frame_surface = pygame.surfarray.make_surface(frame_transposed)
            else:
                pygame.surfarray.blit_array(self.frame_surface, frame_transposed)
            
            # Draw to screen
            self.screen.blit(self.frame_surface, (0, 0))
            
            # Draw border
            pygame.draw.rect(self.screen, self.colors['accent'], 
                           (0, 0, self.display_width, self.display_height), 2)
            
        except Exception as e:
            logger.error(f"Frame draw error: {e}")
    
    def _draw_info_panel(self, detections: List, stats: Dict):
        """Draw information panel"""
        panel_x = self.display_width + 10
        y = 20
        
        # Title
        title = self.font_large.render("SECURITY STATUS", True, self.colors['text'])
        self.screen.blit(title, (panel_x, y))
        y += 40
        
        # Stats
        fps = stats.get('fps', 0)
        fps_color = self.colors['success'] if fps > 10 else self.colors['warning']
        self._draw_stat("FPS:", f"{fps:.1f}", fps_color, panel_x, y)
        y += 30
        
        cpu = stats.get('cpu', 0)
        cpu_color = self.colors['success'] if cpu < 70 else self.colors['warning']
        self._draw_stat("CPU:", f"{cpu:.0f}%", cpu_color, panel_x, y)
        y += 30
        
        mem = stats.get('memory', 0)
        mem_color = self.colors['success'] if mem < 70 else self.colors['warning']
        self._draw_stat("Memory:", f"{mem:.0f}%", mem_color, panel_x, y)
        y += 40
        
        # Detections
        title = self.font_medium.render("DETECTIONS", True, self.colors['text'])
        self.screen.blit(title, (panel_x, y))
        y += 30
        
        humans = len([d for d in detections if d.category == "HUMAN"])
        animals = len([d for d in detections if d.category == "ANIMAL"])
        
        self._draw_stat("Humans:", str(humans), self.colors['text'], panel_x, y)
        y += 25
        self._draw_stat("Animals:", str(animals), self.colors['text'], panel_x, y)
        y += 40
        
        # Threats
        threats = len([d for d in detections if d.threat_level in ["CRITICAL", "HIGH"]])
        threat_color = self.colors['danger'] if threats > 0 else self.colors['success']
        
        title = self.font_medium.render("THREATS", True, self.colors['text'])
        self.screen.blit(title, (panel_x, y))
        y += 30
        
        self._draw_stat("Active:", str(threats), threat_color, panel_x, y)
        y += 40
        
        # Controls
        title = self.font_medium.render("CONTROLS", True, self.colors['text'])
        self.screen.blit(title, (panel_x, y))
        y += 30
        
        controls = ["ESC - Exit", "Q - Quit"]
        for control in controls:
            text = self.font_small.render(control, True, self.colors['text'])
            self.screen.blit(text, (panel_x, y))
            y += 22
    
    def _draw_stat(self, label: str, value: str, color: Tuple, x: int, y: int):
        """Draw a stat line"""
        label_surf = self.font_small.render(label, True, self.colors['text'])
        value_surf = self.font_small.render(value, True, color)
        self.screen.blit(label_surf, (x, y))
        self.screen.blit(value_surf, (x + 120, y))
    
    def handle_events(self) -> bool:
        """Handle events"""
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                return False
            elif event.type == pygame.KEYDOWN:
                if event.key in [pygame.K_ESCAPE, pygame.K_q]:
                    return False
        return True

class OptimizedSecuritySystem:
    """Optimized security system"""
    
    def __init__(self, config: SecurityConfig):
        self.config = config
        self.running = False
        
        # Components
        self.camera = OptimizedCamera(config)
        self.dashboard = FastDashboard(config)
        
        # Load YOLO model
        try:
            logger.info("Loading YOLO model...")
            self.model = YOLO("yolo11n.pt")
            self.model.fuse()  # Fuse layers for speed
            logger.info("YOLO model loaded")
        except Exception as e:
            logger.error(f"Failed to load YOLO: {e}")
            raise
        
        # Detection classes
        self.human_classes = [0]  # person
        self.animal_classes = {
            15: "Cat", 16: "Dog", 17: "Horse", 18: "Sheep", 19: "Cow",
            20: "Elephant", 21: "Bear", 22: "Zebra", 23: "Giraffe"
        }
        self.bird_classes = {14: "Bird"}
        
        # Object heights for distance estimation (meters)
        self.object_heights = {
            'Human': 1.7, 'Cat': 0.23, 'Dog': 0.6, 'Horse': 1.6,
            'Cow': 1.45, 'Elephant': 2.7, 'Bear': 1.0, 'Zebra': 1.3,
            'Giraffe': 4.2, 'Sheep': 0.75, 'Bird': 0.12
        }
        
        # State
        self.last_detection_time = 0
        self.detection_interval = 1.0 / config.DETECTION_FPS
        self.current_detections = []
        self.annotated_frame = None
        
        logger.info("Security system initialized")
    
    def start(self, source: str = "webcam", **kwargs):
        """Start system"""
        logger.info(f"Starting with source: {source}")
        
        # Initialize camera
        if not self.camera.initialize(source, **kwargs):
            logger.error("Camera initialization failed")
            return
        
        # Start capture
        self.camera.start_capture()
        time.sleep(1)  # Wait for camera to stabilize
        
        self.running = True
        
        try:
            while self.running:
                # Get frame
                frame = self.camera.get_frame()
                if frame is None:
                    time.sleep(0.01)
                    continue
                
                # Run detection at lower FPS
                current_time = time.time()
                if current_time - self.last_detection_time >= self.detection_interval:
                    self.current_detections = self._detect_objects(frame)
                    self.annotated_frame = self._draw_detections(frame.copy(), self.current_detections)
                    self.last_detection_time = current_time
                
                # Use annotated frame for display
                display_frame = self.annotated_frame if self.annotated_frame is not None else frame
                
                # Get stats
                stats = {
                    'fps': self.camera.get_fps(),
                    'cpu': psutil.cpu_percent(interval=0),
                    'memory': psutil.virtual_memory().percent
                }
                
                # Update dashboard
                self.dashboard.update(display_frame, self.current_detections, stats)
                
                # Handle events
                if not self.dashboard.handle_events():
                    break
                
        except KeyboardInterrupt:
            logger.info("Interrupted by user")
        except Exception as e:
            logger.error(f"System error: {e}")
            import traceback
            traceback.print_exc()
        finally:
            self.stop()
    
    def _detect_objects(self, frame: np.ndarray) -> List[DetectionEvent]:
        """Run object detection"""
        try:
            # Resize for detection
            if frame.shape[:2] != self.config.DETECTION_RESOLUTION[::-1]:
                detect_frame = cv2.resize(frame, self.config.DETECTION_RESOLUTION)
            else:
                detect_frame = frame
            
            # Run inference
            results = self.model(detect_frame, conf=0.4, iou=0.5, verbose=False, 
                               max_det=self.config.MAX_DETECTIONS)
            
            detections = []
            
            if results and len(results) > 0 and results[0].boxes is not None:
                # Scale factor if we resized
                scale_x = frame.shape[1] / detect_frame.shape[1]
                scale_y = frame.shape[0] / detect_frame.shape[0]
                
                for box in results[0].boxes:
                    cls_id = int(box.cls[0])
                    confidence = float(box.conf[0])
                    
                    # Get object info
                    obj_type, category = self._get_object_info(cls_id)
                    if not obj_type:
                        continue
                    
                    # Check confidence threshold
                    if confidence < self._get_threshold(category):
                        continue
                    
                    # Scale bbox back to original size
                    x1, y1, x2, y2 = map(int, box.xyxy[0])
                    bbox = (int(x1 * scale_x), int(y1 * scale_y),
                           int(x2 * scale_x), int(y2 * scale_y))
                    
                    # Calculate distance
                    distance = self._estimate_distance(bbox, frame.shape, obj_type)
                    
                    # Assess threat
                    threat = self._assess_threat(distance, category)
                    
                    detection = DetectionEvent(
                        timestamp=datetime.now(),
                        object_type=obj_type,
                        category=category,
                        confidence=confidence,
                        bbox=bbox,
                        distance=distance,
                        threat_level=threat
                    )
                    
                    detections.append(detection)
            
            return detections
            
        except Exception as e:
            logger.error(f"Detection error: {e}")
            return []
    
    def _get_object_info(self, cls_id: int) -> Tuple[Optional[str], Optional[str]]:
        """Get object type and category"""
        if cls_id in self.human_classes:
            return "Human", "HUMAN"
        elif cls_id in self.animal_classes:
            return self.animal_classes[cls_id], "ANIMAL"
        elif cls_id in self.bird_classes:
            return self.bird_classes[cls_id], "BIRD"
        return None, None
    
    def _get_threshold(self, category: str) -> float:
        """Get confidence threshold"""
        thresholds = {
            "HUMAN": self.config.HUMAN_CONFIDENCE_THRESHOLD,
            "ANIMAL": self.config.ANIMAL_CONFIDENCE_THRESHOLD,
            "BIRD": self.config.BIRD_CONFIDENCE_THRESHOLD
        }
        return thresholds.get(category, 0.5)
    
    def _estimate_distance(self, bbox: Tuple, frame_shape: Tuple, obj_type: str) -> float:
        """Estimate distance to object"""
        x1, y1, x2, y2 = bbox
        bbox_height = y2 - y1
        frame_height = frame_shape[0]
        height_ratio = bbox_height / frame_height
        
        # Simple distance estimation
        if height_ratio > 0.7:
            return 1.0
        elif height_ratio > 0.5:
            return 2.5
        elif height_ratio > 0.3:
            return 5.0
        elif height_ratio > 0.2:
            return 8.0
        elif height_ratio > 0.1:
            return 12.0
        else:
            return 20.0
    
    def _assess_threat(self, distance: float, category: str) -> str:
        """Assess threat level"""
        if category == "HUMAN":
            if distance < 2:
                return "CRITICAL"
            elif distance < 5:
                return "HIGH"
            elif distance < 10:
                return "MEDIUM"
            else:
                return "LOW"
        elif category == "ANIMAL":
            if distance < 3:
                return "MEDIUM"
            elif distance < 8:
                return "LOW"
        
        return "SAFE"
    
    def _draw_detections(self, frame: np.ndarray, detections: List[DetectionEvent]) -> np.ndarray:
        """Draw detections on frame"""
        for det in detections:
            x1, y1, x2, y2 = det.bbox
            
            # Select color
            if det.threat_level == "CRITICAL":
                color = self.config.THREAT_COLOR_CRITICAL
            elif det.threat_level == "HIGH":
                color = self.config.THREAT_COLOR_HIGH
            elif det.threat_level == "MEDIUM":
                color = self.config.THREAT_COLOR_MEDIUM
            elif det.category == "HUMAN":
                color = self.config.DETECTION_COLOR_HUMAN
            elif det.category == "ANIMAL":
                color = self.config.DETECTION_COLOR_ANIMAL
            else:
                color = self.config.DETECTION_COLOR_BIRD
            
            # Draw box
            thickness = 3 if det.threat_level in ["CRITICAL", "HIGH"] else 2
            cv2.rectangle(frame, (x1, y1), (x2, y2), color, thickness)
            
            # Draw label
            label = f"{det.object_type} {det.distance:.1f}m ({det.confidence:.2f})"
            
            font = cv2.FONT_HERSHEY_SIMPLEX
            font_scale = 0.5
            thickness = 2
            (tw, th), _ = cv2.getTextSize(label, font, font_scale, thickness)
            
            # Background
            cv2.rectangle(frame, (x1, y1 - th - 8), (x1 + tw + 4, y1), color, -1)
            
            # Text
            cv2.putText(frame, label, (x1 + 2, y1 - 4), font, font_scale, (0, 0, 0), thickness)
            
            # Threat label
            if det.threat_level in ["CRITICAL", "HIGH", "MEDIUM"]:
                threat_label = f"THREAT: {det.threat_level}"
                cv2.putText(frame, threat_label, (x1, y2 + 20), font, 0.5, color, 2)
        
        return frame
    
    def stop(self):
        """Stop system"""
        self.running = False
        self.camera.stop()
        self.camera.release()
        if PYGAME_AVAILABLE:
            pygame.quit()
        logger.info("System stopped")
